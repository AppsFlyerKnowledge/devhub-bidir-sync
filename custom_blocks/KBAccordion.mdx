import React, { useState } from 'react';

export const KBAccordion = ({ children, icon, iconColor, title }) => {
  const [isOpen, setIsOpen] = useState(false);

  return (
    <div className="border border-gray-200 dark:border-gray-600 rounded-md overflow-hidden bg-gray-50 dark:bg-gray-800">
      <div
        className={`flex items-center justify-between w-full py-2 px-4 text-base font-medium cursor-pointer hover:bg-gray-100 dark:hover:bg-gray-700 ${isOpen ? 'border-b border-gray-200 dark:border-gray-600' : ''}`}
        onClick={() => setIsOpen(!isOpen)}
      >
        <span className="flex items-center">
          {icon && (
            <i className={`fa-duotone fa-solid ${icon} mr-2`} style={{ color: iconColor }} />
          )}
          <span>{title}</span>
        </span>
        <i className={`fa fa-solid fa-chevron-right text-sm text-gray-500 dark:text-gray-300 mr-2 transition-transform ${isOpen ? 'rotate-90' : ''}`} />
      </div>
      {isOpen && (
        <div className="text-gray-600 dark:text-white px-4 pt-1 pb-2">
          {children}
        </div>
      )}
    </div>
  );
};

<KBAccordion title="Accordion one">
