import React, { useState } from 'react';

export const KBAccordion = ({ children, icon, iconColor, title }) => {
  const [isOpen, setIsOpen] = useState(false);

  return (
    <div
      className="rounded-t-[8px] rounded-b-[5px] overflow-hidden bg-gray-50 dark:bg-gray-800"
      style={{
        borderTop: '1px solid rgba(0, 0, 0, 0.1)',
        borderLeft: '1px solid rgba(0, 0, 0, 0.1)',
        borderRight: '1px solid rgba(0, 0, 0, 0.1)',
        borderBottom: '2px solid #E5E8ED',
      }}
    >
      <div
        className={`flex items-center justify-between w-full pt-[4px] pb-[4px] pl-[15px] pr-[15px] text-base font-medium leading-8 cursor-pointer hover:bg-gray-100 dark:hover:bg-gray-700 ${isOpen ? 'border-b border-gray-200 dark:border-gray-600' : ''}`}
        onClick={() => setIsOpen(!isOpen)}
      >
        <span className="flex items-center">
          {icon && (
            <i className={`fa-duotone fa-solid ${icon} font-black mr-2`} style={{ color: iconColor }} />
          )}
          <span>{title}</span>
        </span>
        <i className={`fa fa-solid fa-chevron-right font-black text-sm text-gray-500 dark:text-gray-300 transition-transform ${isOpen ? 'rotate-90' : ''}`} />
      </div>
      {isOpen && (
        <div className="text-gray-600 dark:text-white pl-[15px] pr-4 pt-1 pb-2">
          {children}
        </div>
      )}
    </div>
  );
};

<KBAccordion title="Accordion one">
Lorem ipsum
</KBAccordion>
